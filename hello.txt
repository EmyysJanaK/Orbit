select email
from Users
group by email -- can not use where bcs where filters before aggregation
having count(email) > 1

delete u1
from Users u1
join Users u2
    on u1.email = u2.email
    and u1.id > u2.id

select id 
from Customers
left join Orders
    on Customers.id = Orders.customer_id
    where order_id is null
